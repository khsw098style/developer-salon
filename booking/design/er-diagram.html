    }

    STAFF {
        uuid id PK
        text name
        text name_en
        enum role
        text bio_role_label
        text bio_comment
        text avatar_image_url
        boolean is_active
        int display_order
    }

    MENUS {
        uuid id PK
        text name
        int price
        int duration_minutes
        boolean is_active
    }

    SITE_FEATURES {
        uuid id PK
        int sort_order
        text title
        text description
        boolean is_active
    }

    SITE_GALLERY_PHOTOS {
        uuid id PK
        enum kind
        text image_url
        text caption
        int sort_order
        boolean is_active
    }

    STAFF_SHIFTS {
        uuid id PK
        uuid staff_id FK
        date date
        boolean is_working
        time start_time
        time end_time
    }

    BUSINESS_DAYS {
        date date PK
        boolean is_open
        time open_time
        time close_time
        time last_reception_time
    }

    RESERVATIONS {
        uuid id PK
        text reservation_number UK
        uuid customer_id FK
        uuid staff_id FK
